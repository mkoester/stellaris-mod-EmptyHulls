# Adds each design once per empire; the flag keeps a deleted design from coming back.
# Scope: country.
ehu_grant_design = {
	if = {
		limit = {
			$WHEN$
			NOT = { has_country_flag = ehu_has_$FLAG$ }
		}
		create_ship_design = { design = "$DESIGN$" ftl = this }
		# keep the tier-1 parts (cheapest); vanilla sets `yes` the same way (grand_archive_effects.txt)
		last_created_design = { set_auto_upgrade_components = no }
		add_ship_design = last_created_design
		set_country_flag = ehu_has_$FLAG$
	}
}

ehu_grant_designs = {
	ehu_grant_design = { FLAG = corvette DESIGN = EHU_S_EMPTY_CORVETTE WHEN = "country_uses_bio_ships = no" }
	ehu_grant_design = { FLAG = mauler DESIGN = EHU_S_EMPTY_MAULER WHEN = "country_uses_bio_ships = yes has_technology = tech_maulers" }
	ehu_grant_design = { FLAG = weaver DESIGN = EHU_M_EMPTY_WEAVER WHEN = "country_uses_bio_ships = yes has_technology = tech_weavers" }
	ehu_grant_design = { FLAG = harbinger DESIGN = EHU_L_EMPTY_HARBINGER WHEN = "country_uses_bio_ships = yes has_technology = tech_harbingers" }
	ehu_grant_design = { FLAG = stinger DESIGN = EHU_XL_EMPTY_STINGER WHEN = "country_uses_bio_ships = yes has_technology = tech_stingers" }
}
